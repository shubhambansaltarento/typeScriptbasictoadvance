any

any essentially disables TypeScript checking for that value.

let data: any;

data = 10;
data = "hello";
data = true;

data.foo.bar.baz(); // TypeScript won't protect you

That's why excessive any is dangerous.

function process(data: any) {
  return data.name.toUpperCase();
}

process(100); // compiles fine, but explodes at runtime

Prefer unknown plus narrowing:

function process(data: unknown) {
  if (
    typeof data === "object" &&
    data !== null &&
    "name" in data
  ) {
    console.log(data.name);
  }
}

any
