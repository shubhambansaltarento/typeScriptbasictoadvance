void

void generally means: this function doesn't return a meaningful value.

function logMessage(message: string): void {
  console.log(message);
}

logMessage("Hello"); // no meaningful result

Compare

function getName(): string {
  return "Shubham";
}

function logName(): void {
  console.log("Shubham");
}

void vs never

function test1(): void {
  console.log("hello");
}
// the function finishes normally

function test2(): never {
  throw new Error("Failed");
}
// the function never finishes normally
